import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        String s = sc.next();

        for (char c : s.toCharArray()) {
            c = Character.toLowerCase(c);

            if (c == 'a' || c == 'e' || c == 'i' ||
                c == 'o' || c == 'u' || c == 'y') {
                continue;
            }

            System.out.print("." + c);
        }
    }
}
