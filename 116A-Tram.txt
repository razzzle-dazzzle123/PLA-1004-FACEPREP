import java.util.*;

public class Tram {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();

        int current = 0;
        int maximum = 0;

        for (int i = 0; i < n; i++) {
            int a = sc.nextInt(); // passengers exiting
            int b = sc.nextInt(); // passengers entering

            current = current - a + b;

            maximum = Math.max(maximum, current);
        }

        System.out.println(maximum);

        sc.close();
    }
}
