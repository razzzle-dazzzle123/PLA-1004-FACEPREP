import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.StringTokenizer;

public class Inversion {
    public static void main(String[] args) throws IOException {
        FastScanner sc = new FastScanner();
        StringBuilder sb = new StringBuilder();
        
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            
            int[] a = new int[n];
            int[] b = new int[n];
            
            boolean allAZero = true;
            boolean allBOne = true;
            boolean identical = true;

            for (int i = 0; i < n; i++) {
                a[i] = sc.nextInt();
                if (a[i] == 1) {
                    allAZero = false;
                }
            }

            for (int i = 0; i < n; i++) {
                b[i] = sc.nextInt();
                if (b[i] == 0) {
                    allBOne = false;
                }
                if (a[i] != b[i]) {
                    identical = false;
                }
            }

            // Case 0: a and b are already equal
            if (identical) {
                sb.append(0).append("\n");
                continue;
            }

            // Case 1: Impossible cases
            if (allAZero || allBOne) {
                sb.append(-1).append("\n");
                continue;
            }

            // Count sum of a_i at indices where a_i != b_i
            int sumMismatchedA = 0;
            for (int i = 0; i < n; i++) {
                if (a[i] != b[i]) {
                    sumMismatchedA += a[i];
                }
            }

            // Case 2: Odd sum -> 1 operation, Even sum -> 2 operations
            if (sumMismatchedA % 2 == 1) {
                sb.append(1).append("\n");
            } else {
                sb.append(2).append("\n");
            }
        }
        
        System.out.print(sb);
    }

    // Fast I/O helper class for large input sizes (n up to 2 * 10^5)
    static class FastScanner {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st;

        String next() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    String line = br.readLine();
                    if (line == null) return null;
                    st = new StringTokenizer(line);
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }
    }
}
