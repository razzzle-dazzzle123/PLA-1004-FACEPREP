import java.io.*;
import java.util.*;

public class Main {

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int t = Integer.parseInt(br.readLine());

        while (t-- > 0) {
            int n = Integer.parseInt(br.readLine());
            String s = br.readLine();

            // Number of transitions in the original string
            int transitions = 0;

            for (int i = 1; i < n; i++) {
                if (s.charAt(i) != s.charAt(i - 1)) {
                    transitions++;
                }
            }

            int ans = Integer.MAX_VALUE;

            // Delete s[i], cannot delete first or last character
            for (int i = 1; i < n - 1; i++) {

                int newTransitions = transitions;

                // Remove the two old edges
                if (s.charAt(i - 1) != s.charAt(i)) {
                    newTransitions--;
                }

                if (s.charAt(i) != s.charAt(i + 1)) {
                    newTransitions--;
                }

                // Add the new edge created after deletion
                if (s.charAt(i - 1) != s.charAt(i + 1)) {
                    newTransitions++;
                }

                int compressedLength = newTransitions + 1;

                ans = Math.min(ans, compressedLength);
            }

            System.out.println(ans);
        }
    }
}
