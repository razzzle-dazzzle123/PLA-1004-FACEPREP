import java.io.*;
import java.util.*;

public class DominoTiles {

    static final long MOD = 998244353;

    static long count(String s, int start) {
        // start = 0 -> positions 0,2,4,...
        // start = 1 -> positions 1,3,5,...

        long ways = 0;

        // Try both possible starting values
        for (int first = 0; first <= 1; first++) {
            boolean possible = true;

            for (int i = start, pos = 0; i < s.length(); i += 2, pos++) {
                int expected = first ^ (pos % 2);

                if (s.charAt(i) != '?' &&
                    s.charAt(i) - '0' != expected) {
                    possible = false;
                    break;
                }
            }

            if (possible) {
                ways++;
            }
        }

        return ways;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int t = Integer.parseInt(br.readLine());

        while (t-- > 0) {
            int n = Integer.parseInt(br.readLine());
            String s = br.readLine();

            long oddWays = count(s, 0);
            long evenWays = count(s, 1);

            long ans = (oddWays * evenWays) % MOD;

            System.out.println(ans);
        }
    }
}
